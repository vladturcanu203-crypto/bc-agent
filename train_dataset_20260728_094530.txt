Domanda: Come si crea una query con colonne calcolate?
File: BLDOBJQRY_COMUNE_ELENCO_DEFINIZIONE - Copia.BC
Contesto:
'@PGQUERY BLDOBJQRY_COMUNE_ELENCO_DEFINIZIONE STRICT[4] ' Query di elenco su BLDCOMUNE - Definizione query
  
    '@SBP Query[BCQUERY] [INOUT]                      _ ' Oggetto Query
         Multisel[TIPO[BCBOOL]]              [IN]    _ ' Multiselezione (0-No;1-Si)
         CodPreselezionati[STRING]           [IN]    _ ' Elenco codici preselezionati
         Modalita[TIPO[BCTD_MODALITAELENCO]] [IN]    _ ' Modalit dell'elenco
         Filtri[CLASSE[BLD_COMUNE_FILTRI]]   [INOUT] 

   '@DEFQUERY QUERY[Query] NOME["OBJQRY_BLD_COMUNE_ELENCO_DEFINIZIONE_IdQueryElenco"]
   
      '@FROM TABELLA[BLDCOMUNE]
   
      '@COLUMN ESPR[BLDCOMUNE.COD AS [Comune]]         _
               ESPR[BLDCOMUNE.CODPR AS [CodProvincia]] _
               ESPR[BLDCOMUNE.CODREG AS [CodRegione]]  _
               ESPR[BLDCOMUNE.CODST AS [CodStato]]     _
               ESPR[BLDCOMUNE.CAP AS [CAP]]
                   
      If Multisel = 1 Then
   
         '@COMPUTEDCOLUMN ESPR[If INLISTSTR(BLDFAMIGLIA.COD, CodPreSelezionati)>0 Then 1 AS [ZSEL] FMT[NUMERICO[1]] ]
   
      EndIf   
      
      If Filtri.CodComune <> "" Then
      
         '@WHERE ESPR[BLDCOMUNE.COD = Filtri.CodComune]
      
      Else
      
         If Modalita = BCTD_MODALITAELENCO.Ricerca And Filtri.CodStato <> "" Then 
   
            '@WHERE ESPR[BLDCOMUNE.CODST = Filtri.CodStato]
      
         Else
      
            '@WHERE ESPR[BLDCOMUNE.COD <> ""]
      
         EndIf 
         
      EndIf

   '@ENDQUERY

End

'@PGQUERY BLDOBJQRY_QUERYBASE_DEFINIZIONE STRICT[4] ' Esercitazione query base - Definizione query

   '@SBP QueryObject[BCQUERY] [INOUT]

   '@DEFQUERY QUERY[QueryObject] NOME["BLDOBJQRY_QUERYBASE_DEFINIZIONE"]
   
      ' Definizione della tabella di riferimento della query
      ' Tabella ARTICOLI
      '@FROM TABELLA[BLDART]
      
      ' Definizione dei campi della query
      ' Una sola DEFQUERYFIELD con tanti ESPR quanti sono i campi
      '@COLUMN ESPR[BLDART.COD       AS [CodiceArticolo]]      _ ' Codice articolo
               ESPR[BLDART.DES       AS [Descrizione]]         _ ' Descrizione
               ESPR[BLDART.CODFAM    AS [Famiglia]]            _ ' Codice famiglia
               ESPR[BLDART.PREZZO    AS [PrezzoUnitario]]      _ ' Prezzo unitario di listino
               ESPR[BLDART.SCONTO    AS [ScontoArticolo]]      _ ' Sconto articolo
               ESPR[BLDART.QUA       AS [QuantitaMagazzino]]     ' Quantit in magazzino
   
      ' Filtri (Where)
      ' Applicazione del filtro: Selezionare tutti i dati DOVE il prezzo  superiore a 30
      '@WHERE ESPR[BLDART.PREZZO>30]
      
      ' Ordinamento
      ' Definizione ordinamento ASCENDENTE per DESCRIZIONE
      '@ORDERBY IND[1] ESPR[BLDART.DES/ASC]

   '@ENDQUERY

End

Domanda: Spiega l'uso di JOIN nelle query BC
File: BLDOBJQRY_CLIFOR_ELENCO_DEFINIZIONE - Copia.BC
Contesto:
'@PGQUERY BLDOBJQRY_CLIFOR_ELENCO_DEFINIZIONE STRICT[4] ' Query di elenco su BLDCLIFOR - Definizione query

 '@SBP Query[BCQUERY]                    [INOUT] _ ' Oggetto query
         Multisel[TIPO[BCBOOL]]            [IN]    _ ' Multiselezione (0-No;1-Si)
         CodPreselezionati[STRING]         [IN]    _ ' Elenco codici preselezionati
         Filtri[CLASSE[BLD_CLIFOR_FILTRI]] [INOUT]   ' Filtri

   '@DEFQUERY QUERY[Query] NOME["OBJQRY_BLD_CLIFOR_ELENCO_DEFINIZIONE_IdQueryElenco"]

      '@FROM TABELLA[BLDCLIFOR]
   
      '@COLUMN ESPR[BLDCLIFOR.COD  AS [Codice]] _
               ESPR[BLDCLIFOR.RS1  AS [RS1]]    _
               ESPR[BLDCLIFOR.PIVA AS [PIVA]]
      
      '@COMPUTEDCOLUMN ESPR[DECODDES$(BLDCLIFOR.TIPANA) AS[Tipana] FMT[TESTO[50]]]
      '@COMPUTEDCOLUMN ESPR[DECODDES$(BLDCLIFOR.STANA)  AS[Stana]  FMT[TESTO[50]]] 
      '@COMPUTEDCOLUMN ESPR[DECODDES$(BLDCLIFOR.TIPSOG) AS[Tipsog] FMT[TESTO[50]]]
      
      ' Se  prevista la multiselezione va aggiunto il campo calcolato ZSEL sui codici preselezionati
      If Multisel = 1 Then
   
         '@COMPUTEDCOLUMN ESPR[If INLISTSTR(BLDCLIFOR.COD, CodPreSelezionati) > 0 Then 1 AS [ZSEL] FMT[NUMERICO[1]]]
   
      EndIf
      
      If Filtri.CodCliFor <> 0 Then
         
         '@WHERE ESPR[BLDCLIFOR.COD = Filtri.CodCliFor]
         
      Else
      
         If Filtri.TipoFiltro <> -1 AND Filtri.TipoFiltro <> 2 Then
      
            '@WHERE ESPR[BLDCLIFOR.TIPANA = Filtri.TipoFiltro]
         
         EndIf
         
         If Filtri.TipoFiltro = 2 Then
         
            '@WHERE ESPR[BLDCLIFOR.STANA = Filtri.TipoFiltro]
         
         EndIf
         
      EndIf
      
   '@ENDQUERY

End

      '@DEFQUERYCTE QUERY[ParteFissa] ALIAS["Organigramma"] PARTE[#FISSA]
      
         '@FROM TABELLA[BLDIMPIEGATI]
         
         '@COLUMN ESPR[BLDIMPIEGATI.IDImpiegato AS [IdPadre]] 
         '@COMPUTEDCOLUMN ESPR[BLDIMPIEGATI.Nome +" " + BLDIMPIEGATI.Cognome AS [Nominativo] FMT[TESTO[80]]] 
         '@COMPUTEDCOLUMN ESPR[1 AS [Livello] FMT[NUMERICO[2]]] 
         '@COMPUTEDCOLUMN ESPR[FNTRSD$(BLDIMPIEGATI.Nome) + " " + FNTRSD$(BLDIMPIEGATI.Cognome) AS [Posizione] FMT[TESTO[255]]]  
         '@COMPUTEDCOLUMN ESPR[FNS$(BLDIMPIEGATI.IDImpiegato) AS [Sort] FMT[TESTO[255]]]    
         
         '@WHERE ESPR[BLDIMPIEGATI.IDResponsabile=0]
      
      '@ENDQUERY 
      
      '@DEFQUERYCTE QUERY[ParteRicorsiva] ALIAS["Organigramma"] PARTE[#RICORSIVA]
      
         '@FROM TABELLA[BLDIMPIEGATI]
         '@JOIN TABELLA[[Organigramma]] TABELLAPADRE[BLDIMPIEGATI] OPERATOREJOIN["INNER"] REL[BLDIMPIEGATI.IDResponsabile=[Organigramma].IdPadre] 
         
         '@COLUMN ESPR[BLDIMPIEGATI.IDImpiegato AS [IdPadre]]

Domanda: Spiega l'uso di JOIN nelle query BC
File: BLDOBJQRY_ART_ELENCO_DEFINIZIONE.BC
Contesto:
'@PGQUERY BLDOBJQRY_ART_ELENCO_DEFINIZIONE STRICT[4] ' Query di elenco su BLDART - Definizione query

   '@SBP Query[BCQUERY]                   [INOUT] _ ' Oggetto query
         Multisel[TIPO[BCBOOL]]           [IN]    _ ' Abilita multiselezione
         CodPreselezionati[STRING]        [IN]    _ ' Elenco codici pre selezionati
         Filtri[CLASSE[BLD_ART_FILTRI]]   [INOUT]   ' Filtro sul campo chiave per l'aggiornamento della griglia

   '@DEFQUERY QUERY[Query] NOME["OBJQRY_BLD_ART_ELENCO_DEFINIZIONE_IdQueryElenco"]

      '@FROM TABELLA[BLDART]
      
      '@JOIN TABELLA[BLDFAMIGLIA] TABELLAPADRE[BLDART] _
             REL[BLDFAMIGLIA.COD = BLDART.CODFAM] OPERATOREJOIN["LEFT"]
             
      '@COLUMN ESPR[BLDART.COD            AS [CodiceArticolo]] _
               ESPR[BLDART.DES            AS [Descrizione]]    _
               ESPR[BLDART.ACQUISTO       AS [Acquisto]]       _
               ESPR[BLDART.VENDITA        AS [Vendita]]        _
               ESPR[BLDART.GESVAR         AS [GestVariazione]] _
               ESPR[BLDART.FlGesMatricole AS [GestMatricole]]  _
               ESPR[BLDART.COSTO          AS [Costo]]          _
               ESPR[BLDART.PREZZO         AS [Prezzo]]         _
               ESPR[BLDART.SCONTO         AS [Sconto]]         _
               ESPR[BLDART.CODFAM         AS [CodiceFamiglia]] _
               ESPR[BLDFAMIGLIA.DES       AS [DescrizioneFam]]
               
      '@COMPUTEDCOLUMN ESPR[DECODDES$(BLDART.STATO) AS[Stato] FMT[TESTO[50]]] _
                       ESPR[DECODDES$(BLDART.TIPO)  AS[Tipo]  FMT[TESTO[50]]]
                       
      ' Se  prevista la multiselezione va aggiunto il campo calcolato ZSEL sui codici preselezionati
      If Multisel = #True Then

         '@COMPUTEDCOLUMN ESPR[If INLISTSTR(BLDART.COD, CodPreSelezionati) > 0 Then 1 AS [ZSEL] FMT[NUMERICO[1]]]

      EndIf
      
      ' Escludo i filtri se sto aggiornando la query
      If Filtri.CodArt = "" Then
   
         If Filtri.CodMacro <> "" Then
         
           '@WHERE ESPR[INLISTSTR(BLDFAMIGLIA.CODMACRO, Filtri.CodMacro) > 0]
           
         EndIf
         
         If Filtri.CodFamSelezionati <> "" Then
         
           '@WHERE ESPR[INLISTSTR(BLDART.CODFAM, Filtri.CodFamSelezionati) > 0]
           
         EndIf
         
         If Filtri.Acquisti <> 2 Then
         
            '@DEFQUERYWHERE ESPR[BLDART.ACQUISTO = Filtri.Acquisti]
            
         EndIf
         
         If Filtri.Vendite <> 2 Then
         
            '@WHERE ESPR[BLDART.VENDITA = Filtri.Vendite]
            
         EndIf
         
         If Filtri.DesArt > "" Then
         
            '@WHERE ESPR[MATCH(BLDART.DES, "%" + FNTRSD$(Filtri.DesArt) + "%") > 0]
            
         EndIf
         
      Else
      
         '@WHERE ESPR[BLDART.COD = Filtri.CodArt]
      
      EndIf
      
   '@ENDQUERY

End

Domanda: Come si crea una query con colonne calcolate?
File: BLDOBJQRY_ART_ELENCO_DEFINIZIONE.BC
Contesto:
'@PGQUERY BLDOBJQRY_ART_ELENCO_DEFINIZIONE STRICT[4] ' Query di elenco su BLDART - Definizione query

   '@SBP Query[BCQUERY]                   [INOUT] _ ' Oggetto query
         Multisel[TIPO[BCBOOL]]           [IN]    _ ' Abilita multiselezione
         CodPreselezionati[STRING]        [IN]    _ ' Elenco codici pre selezionati
         Filtri[CLASSE[BLD_ART_FILTRI]]   [INOUT]   ' Filtro sul campo chiave per l'aggiornamento della griglia

   '@DEFQUERY QUERY[Query] NOME["OBJQRY_BLD_ART_ELENCO_DEFINIZIONE_IdQueryElenco"]

      '@FROM TABELLA[BLDART]
      
      '@JOIN TABELLA[BLDFAMIGLIA] TABELLAPADRE[BLDART] _
             REL[BLDFAMIGLIA.COD = BLDART.CODFAM] OPERATOREJOIN["LEFT"]
             
      '@COLUMN ESPR[BLDART.COD            AS [CodiceArticolo]] _
               ESPR[BLDART.DES            AS [Descrizione]]    _
               ESPR[BLDART.ACQUISTO       AS [Acquisto]]       _
               ESPR[BLDART.VENDITA        AS [Vendita]]        _
               ESPR[BLDART.GESVAR         AS [GestVariazione]] _
               ESPR[BLDART.FlGesMatricole AS [GestMatricole]]  _
               ESPR[BLDART.COSTO          AS [Costo]]          _
               ESPR[BLDART.PREZZO         AS [Prezzo]]         _
               ESPR[BLDART.SCONTO         AS [Sconto]]         _
               ESPR[BLDART.CODFAM         AS [CodiceFamiglia]] _
               ESPR[BLDFAMIGLIA.DES       AS [DescrizioneFam]]
               
      '@COMPUTEDCOLUMN ESPR[DECODDES$(BLDART.STATO) AS[Stato] FMT[TESTO[50]]] _
                       ESPR[DECODDES$(BLDART.TIPO)  AS[Tipo]  FMT[TESTO[50]]]
                       
      ' Se  prevista la multiselezione va aggiunto il campo calcolato ZSEL sui codici preselezionati
      If Multisel = #True Then

         '@COMPUTEDCOLUMN ESPR[If INLISTSTR(BLDART.COD, CodPreSelezionati) > 0 Then 1 AS [ZSEL] FMT[NUMERICO[1]]]

      EndIf
      
      ' Escludo i filtri se sto aggiornando la query
      If Filtri.CodArt = "" Then
   
         If Filtri.CodMacro <> "" Then
         
           '@WHERE ESPR[INLISTSTR(BLDFAMIGLIA.CODMACRO, Filtri.CodMacro) > 0]
           
         EndIf
         
         If Filtri.CodFamSelezionati <> "" Then
         
           '@WHERE ESPR[INLISTSTR(BLDART.CODFAM, Filtri.CodFamSelezionati) > 0]
           
         EndIf
         
         If Filtri.Acquisti <> 2 Then
         
            '@DEFQUERYWHERE ESPR[BLDART.ACQUISTO = Filtri.Acquisti]
            
         EndIf
         
         If Filtri.Vendite <> 2 Then
         
            '@WHERE ESPR[BLDART.VENDITA = Filtri.Vendite]
            
         EndIf
         
         If Filtri.DesArt > "" Then
         
            '@WHERE ESPR[MATCH(BLDART.DES, "%" + FNTRSD$(Filtri.DesArt) + "%") > 0]
            
         EndIf
         
      Else
      
         '@WHERE ESPR[BLDART.COD = Filtri.CodArt]
      
      EndIf
      
   '@ENDQUERY

End

'@PG SUB06_LETTURA      ' Esercitazione 6: Lettura articoli
